name: Update Predictions

on:
  schedule:
    # প্রতিদিন UTC সকাল ৬টায় (বাংলাদেশ সময় দুপুর ১২টা) — চাইলে সময় পরিবর্তন করুন
    - cron: "0 6 * * *"
  workflow_dispatch: {}   # ম্যানুয়ালি "Run workflow" বাটন দিয়েও চালানো যাবে

permissions:
  contents: write

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: pip install requests

      - name: Generate predictions.json
        env:
          FOOTBALL_DATA_API_KEY: ${{ secrets.FOOTBALL_DATA_API_KEY }}
        run: python scripts/generate_predictions.py

      - name: Commit updated data
        run: |
          git config user.name "predictions-bot"
          git config user.email "actions@github.com"
          git add data/predictions.json
          git diff --staged --quiet || git commit -m "Update predictions [auto]"
          git push
